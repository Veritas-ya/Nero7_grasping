cmake_minimum_required(VERSION 3.10)
# Ubuntu 22.04's Gazebo dependencies declare older policy levels. Support the
# user's CMake 4 installation without changing those system packages.
if(CMAKE_VERSION VERSION_GREATER_EQUAL "4.0")
  set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
endif()
project(nero7_sim)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(gazebo REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(gazebo_msgs REQUIRED)
add_library(nero7_scene SHARED plugins/scene.cpp)
target_compile_features(nero7_scene PUBLIC cxx_std_17)
target_include_directories(nero7_scene PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_libraries(nero7_scene ${GAZEBO_LIBRARIES})
ament_target_dependencies(nero7_scene gazebo_ros rclcpp sensor_msgs geometry_msgs std_msgs std_srvs gazebo_msgs)
ament_python_install_package(${PROJECT_NAME})
install(TARGETS nero7_scene LIBRARY DESTINATION lib)
install(PROGRAMS scripts/sim_driver scripts/sim_scenario scripts/sim_validate scripts/sim_camera_sync DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
